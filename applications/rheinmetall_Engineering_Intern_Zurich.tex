\projectentry{LoRA Fine-Tuning Engine}{Python, PyTorch, from-scratch transformer}
\begin{itemize}
  \item Hand-written decoder-only transformer and LoRA layer on raw PyTorch autograd (no PEFT/Trainer); LoRA math checked against hand-computed values, plus a test that frozen base weights stay non-trainable.
  \item Measured 3-way comparison (no fine-tune / full / LoRA $r{=}8$): LoRA trains only \textbf{3.85\%} of parameters (24.6K vs 614.8K) and forgets less (generic val.\ loss 2.20 vs 3.08 for full fine-tuning) but adapts less to the new domain (2.33 vs 0.80); 13 tests.
\end{itemize}

\projectentry{IIoT Predictive Maintenance}{Rust, tokio, anomaly detection (EWMA/CUSUM)}
\begin{itemize}
  \item Async telemetry pipeline (\texttt{tokio} mpsc) fuses EWMA + CUSUM control charts across 3 sensor channels to detect developing bearing-wear signatures, with linear-trend remaining-useful-life estimation.
  \item Sensor-fusion tuning (2-of-3 channels + widened control limits) cut false positives from a 0.41--0.50 precision baseline to \textbf{0/10 across 10 seeds}; precision \textbf{1.0000}, recall 0.9642, F1 0.9818, 43-sample latency; \textbf{1.6\,M samples/s}; 26 tests, 0 failures.
\end{itemize}

\projectentry{ICS Intrusion Detection (IEC 60870-5-104)}{C++20, intrusion/anomaly detection, explainable rules}
\begin{itemize}

\end{itemize}
